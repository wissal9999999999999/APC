       this .setText("Bonjour");

 // gestion du bouton "ajout"
 private class AjoutListener implements ActionListener {
   public void actionPerformed(ActionEvent e) {
     String contenuChampsPersonne = nomPersonne.getText().trim();
     // si on rentre un nom vide : Beeeeeeeeeeeeeeeeeeeeeeeeeep
      if (contenuChampsPersonne.equals("")) {
        Toolkit.getDefaultToolkit().beep();
        return;
     }

         int index = list.getSelectedIndex();
         int size = listModel.getSize();

         // si on ajoute en fin de liste ou dans une liste vide
         if ( index == -1 || (index+1 == size) ) {
           listModel.addElement(contenuChampsPersonne);
           list.setSelectedIndex(size);
         }
         // si on ajoute en milieu de liste
         else {
           listModel.insertElementAt(contenuChampsPersonne, index+1);
           list.setSelectedIndex(index+1);
         }
     }
 }

Custom Painting
Objectif du Custom Painting (Personnalisation)
      ajout de dessins à un composant existant
      création d’un nouveau composant

Comment personnaliser ?
      spécialisation d’un composant existant (JButton, JLabel, ...)
      création d’un nouveau composant (conseil : à partir d’un JPanel)
Avec redéfinition de :

                     public void paintComponent(Graphics)

Remarque :
Si vous souhaitez conserver le dessin original d’un composant, vous devez
commencer par appeler super.paintComponent(Graphics)

JLabelBarre.java
import javax.swing.*;
import javax.swing.border.*;
import java.awt.*;

public class JLabelBarre extends JLabel{

    public JLabelBarre(String titre, int horizontalAlignment) {
      super(titre, horizontalAlignment);
    }
    public void paintComponent(Graphics g) {
      super.paintComponent(g);

        Insets insets = getInsets();

        g.setColor(Color.black);
        g.drawLine(insets.left, insets.top,
                  getWidth()-insets.right-1, getHeight()-insets.bottom-1);
        g.drawLine(insets.left, getHeight()-insets.bottom-1,
                  getWidth()-insets.right-1, insets.top);
    }
}
