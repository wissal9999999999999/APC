--- PAGE 20 ---
La classe Observable                                   2/2

Méthodes de changement d'état
   ●   protected synchronized void setChanged()
   Spécifie que l'objet a changé d'état
   ●   protected synchronized void clearChanged()
   Spécifie que l'objet est inchangé, ou qu'il a déjà notifié ses Observers
   Méthode appelée automatiquement par notifyObservers()
   ●   public synchronized boolean hasChanged()
   Teste si l'objet a changé d'état


Notification
   ●   public void notifyObservers()
   Informe les Observers que l'objet a changé d'état
   ●   public void notifyObservers(Object arg)
   Informe les Observers que l'objet a changé d'état,
   et transmet l'objet arg aux Observers

--- PAGE 23 ---
Exemple Observable                      2/3



class Nous extends Observable {
   private int money=0;
   void earnMoney(int val) {
       if (val!=0) {
          money += val;
          System.out.println("J'ai gagné "+val+"$");
          setChanged();
          notifyObservers();
       }
   }
   int getMoney() {
       return money;
   }
   int giveMoney() {
       int tmp = money;
       money = 0;
       return tmp;
   }
}

--- PAGE 26 ---
Un Modèle de température
          ● Encapsulation d'un double = température en Farenheit

          ● Le modèle doit être observable

public class TemperatureModel extends Observable {
    private double temperatureFarenheit=32;
    public double getFarenheit() {
        return temperatureFarenheit;
    }
    public double getCelsius() {
        return (temperatureFarenheit - 32.0) * 5.0 / 9.0;
    }
    public void setFarenheit(double _temperatureFarenheit) {
    // verification du changement de valeur
        if (temperatureFarenheit != _temperatureFarenheit) {
            temperatureFarenheit = _temperatureFarenheit;
            setChanged();
            notifyObservers();
        }
    }
    public void setCelsius(double temperatureCelsius) {
        setFarenheit(temperatureCelsius*9.0/5.0 + 32.0);
    }
}

Une simple Vue
public class FarenheitGauge extends JFrame implements Observer {
    private TemperatureModel model;
    private JProgressBar gauge;

   public FarenheitGauge(TemperatureModel _model) {
       super("Farenheit gauge");
       model = _model;
       model.addObserver(this);

       gauge = new JProgressBar(JProgressBar.VERTICAL,-20,150);
       gauge.setStringPainted(true);
       gauge.setBorderPainted(true);
       getContentPane().setLayout(new FlowLayout());
       getContentPane().add(gauge);
       update(model, null);

       setDefaultCloseOperation(EXIT_ON_CLOSE);
       pack();
       setVisible(true);
   }

   public void update(Observable observable, Object o) {
       int val = Math.round((float)model.getFarenheit());
       gauge.setString(val+"F");
       gauge.setValue(val);
   }                                                               28
